# Update-Project 📦🔍

A Python utility and GUI tool designed to aggregate entire codebases into structured context files for AI audits, LLM code reviews, and clean distribution archives.

---

[🇧🇷 Ler em Português](#-visão-geral-em-português) | [🇺🇸 English Overview](#-english-overview)

---

## 🇺🇸 English Overview

### The Problem
When working with Large Language Models (ChatGPT, Claude, Gemini, DeepSeek, etc.) on medium-to-large projects, AI models frequently suffer from **context fragmentation**. Because they cannot inspect hundreds of scattered files simultaneously without losing attention, code audits often lead to hallucinations, false assumptions, or missed architectural contracts.

### The Solution
**Update-Project** scans your project, applies clean exclusion rules (ignoring vendor folders, cache, and build artifacts), and generates:
1. **A Structured Context File (`{project_name}.txt` and `project.txt`)**: Contains the complete directory tree followed by the full source code of each file, cleanly demarcated with file path headers.
2. **A Clean Project Archive (`{project_name}.zip`)**: A ready-to-deploy zip archive containing only valid source files.

You can then attach `{project_name}.txt` directly to any AI tool to perform complete, zero-hallucination code audits and architectural reviews.


import os
import zipfile
import tempfile
import shutil
from typing import List, Callable
from core.filters import FileFilter

class Compressor:
    def compress(self, root_path: str, selected_paths: set, file_filter: FileFilter, output_zip_path: str, progress_callback: Callable[[int, int, str], None] = None):
        """
        Cria o arquivo ZIP.
        """
        # Passo 1: Filtrar arquivos válidos com base na lista de selecionados
        files_to_zip = []
        for root, dirs, files in os.walk(root_path):
            for file in files:
                file_path = os.path.join(root, file)
                
                # Feedback de preparação
                if progress_callback:
                    progress_callback(0, 0, f"Preparando ZIP: {file}")
                
                # Guarda de trânsito: Verifica exceções
                if file_filter.should_ignore_file(file):
                    continue
                
                # Só compacta se o arquivo em si estiver na lista de selecionados.
                if file_path in selected_paths:
                    arcname = os.path.relpath(file_path, root_path)
                    files_to_zip.append((file_path, arcname))
        
        total_zip = len(files_to_zip)
        zip_count = 0

        if total_zip == 0:
            if progress_callback:
                progress_callback(0, 0, "Nenhum arquivo selecionado para compactação.")
            return

        import tempfile
        import shutil

        temp_dir = tempfile.gettempdir()
        project_name = os.path.basename(os.path.normpath(root_path))
        temp_zip_base = os.path.join(temp_dir, f"{project_name}_temp.zip")

        if os.path.exists(output_zip_path):
            os.remove(output_zip_path)

        with zipfile.ZipFile(temp_zip_base, 'w', zipfile.ZIP_DEFLATED) as zipf:
            for file_path, arcname in files_to_zip:
                # Evitar falhas se o arquivo de destino (zip antigo) tiver sido marcado
                # e removido pelo os.remove acima.
                if os.path.abspath(file_path) == os.path.abspath(output_zip_path):
                    continue
                
                zip_count += 1
                if progress_callback:
                    progress_callback(zip_count, total_zip, arcname)
                zipf.write(file_path, arcname)

        shutil.move(temp_zip_base, output_zip_path)

import fnmatch
import os

class FileFilter:
    def __init__(self, ignore_patterns: list, default_ignore_files: set, default_ignore_dirs: set, allowed_extensions: tuple):
        self.ignore_patterns = ignore_patterns if ignore_patterns else []
        self.default_ignore_files = default_ignore_files
        self.default_ignore_dirs = default_ignore_dirs
        self.allowed_extensions = allowed_extensions

    def should_ignore_file(self, filename: str, is_test: bool = False) -> bool:
        if filename in self.default_ignore_files or filename.endswith('.tmp'):
            return True
        if is_test:
            return True
        for pattern in self.ignore_patterns:
            if fnmatch.fnmatch(filename.lower(), pattern.lower()):
                return True
        return False

    def should_ignore_dir(self, dirname: str) -> bool:
        if dirname in self.default_ignore_dirs:
            return True
        for pattern in self.ignore_patterns:
            if fnmatch.fnmatch(dirname.lower(), pattern.lower()):
                return True
        return False

    def is_allowed_for_report(self, filepath: str) -> bool:
        if not os.path.isfile(filepath):
            return False
        if not filepath.endswith(self.allowed_extensions):
            return False
        if filepath.endswith('.min.css') or filepath.endswith('.min.js'):
            return False
        
        filename = os.path.basename(filepath)
        for pattern in self.ignore_patterns:
            if fnmatch.fnmatch(filename.lower(), pattern.lower()):
                return False
        return True

import os
from typing import List, Optional

class FileNode:
    """Representa um arquivo ou diretório na árvore escaneada."""
    def __init__(self, name: str, path: str, is_dir: bool):
        self.name = name
        self.path = path
        self.is_dir = is_dir
        self.children: List['FileNode'] = []
        # Para diretórios, indica se ele foi completamente ignorado nas configurações
        self.is_ignored = False 

    def add_child(self, child: 'FileNode'):
        self.children.append(child)

    def sort_children(self):
        """Ordena: Pastas primeiro, depois arquivos alfabeticamente."""
        self.children.sort(key=lambda x: (not x.is_dir, x.name.lower()))
        for child in self.children:
            if child.is_dir:
                child.sort_children()

import os
from datetime import datetime
from typing import Callable, List
from core.tree_formatter import TreeFormatter
from core.filters import FileFilter

class ReportGenerator:
    def generate(self, root_path: str, selected_paths: set, file_filter: FileFilter, output_file: str, progress_callback: Callable[[int, int, str], None] = None):
        tree_str = TreeFormatter.get_tree_string(root_path, selected_paths, progress_callback)
        
        files_to_read = []
        for p in selected_paths:
            if file_filter.is_allowed_for_report(p):
                # Evita ler o próprio arquivo de saída se já existir
                if os.path.abspath(p) == os.path.abspath(output_file):
                    continue
                files_to_read.append(p)
        
        files_to_read.sort()
        total_files = len(files_to_read)
        project_name = os.path.basename(os.path.normpath(root_path))
        now_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        
        with open(output_file, 'w', encoding='utf-8') as f_out:
            # Cabeçalho estruturado para LLMs / IAs e auditoria
            f_out.write("/* " + "="*70 + " */\n")
            f_out.write(f"/* PROJETO: {project_name}\n")
            f_out.write(f"/* DATA DE GERAÇÃO: {now_str}\n")
            f_out.write(f"/* TOTAL DE ARQUIVOS CONSOLIDADOS: {total_files}\n")
            f_out.write("/* FERRAMENTA: Update-Project (Codebase Context Packer)\n")
            f_out.write("/* FINALIDADE: Auditoria de código, revisão estática e contexto para IAs (LLMs)\n")
            f_out.write("/* " + "="*70 + " */\n\n")
            
            # Árvore de Diretórios
            f_out.write("/* " + "-"*70 + " */\n")
            f_out.write("/* ESTRUTURA DO PROJETO (ÁRVORE DE ARQUIVOS)\n")
            f_out.write("/* " + "-"*70 + " */\n\n")
            f_out.write(tree_str + "\n\n")
            
            # Código dos arquivos
            f_out.write("/* " + "="*70 + " */\n")
            f_out.write("/* CONTEÚDO DOS ARQUIVOS DO PROJETO\n")
            f_out.write("/* " + "="*70 + " */\n\n")
            
            for i, f_path in enumerate(files_to_read, 1):
                relative_path = os.path.relpath(f_path, root_path).replace("\\", "/")
                if progress_callback:
                    progress_callback(i, total_files, f"Lendo arquivo: {relative_path}")
                
                f_out.write("/* " + "-"*70 + " */\n")
                f_out.write(f"// Arquivo: {project_name}/{relative_path}\n")
                f_out.write("/* " + "-"*70 + " */\n\n")
                
                try:
                    with open(f_path, 'r', encoding='utf-8', errors='ignore') as f_in:
                        f_out.write(f_in.read() + "\n\n")
                except Exception as e:
                    f_out.write(f"// [Erro ao ler arquivo: {e}]\n\n")

import os
from core.models import FileNode
from core.filters import FileFilter

class Scanner:
    def __init__(self, root_path: str, file_filter: FileFilter, is_test_file_fn: callable):
        self.root_path = root_path
        self.total_dirs = 0
        self.total_files = 0
        self.file_filter = file_filter
        self.is_test_file_fn = is_test_file_fn

    def scan(self, progress_callback=None, item_callback=None) -> FileNode:
        root_name = os.path.basename(os.path.normpath(self.root_path))
        root_node = FileNode(root_name, self.root_path, is_dir=True)
        if item_callback:
            # Envia a raiz com parent_path Vazio
            item_callback("", self.root_path, True, root_name)
            
        self._build_tree(root_node, progress_callback, item_callback)
        root_node.sort_children()
        return root_node

    def _build_tree(self, node: FileNode, progress_callback=None, item_callback=None):
        try:
            items = os.listdir(node.path)
        except PermissionError:
            return

        for item in items:
            full_path = os.path.join(node.path, item)
            is_dir = os.path.isdir(full_path)
            
            if is_dir:
                if self.file_filter.should_ignore_dir(item):
                    continue
            else:
                if self.file_filter.should_ignore_file(item, self.is_test_file_fn(item)):
                    continue

            child_node = FileNode(item, full_path, is_dir)
            node.add_child(child_node)

            if is_dir:
                self.total_dirs += 1
                if progress_callback:
                    progress_callback(f"Escaneando pasta: {item}")
                if item_callback:
                    item_callback(node.path, full_path, True, item)
                self._build_tree(child_node, progress_callback, item_callback)
            else:
                self.total_files += 1
                if progress_callback:
                    if self.total_files % 10 == 0:
                        progress_callback(f"Encontrados {self.total_files} arquivos...")
                if item_callback:
                    item_callback(node.path, full_path, False, item)

import os
import shutil
from typing import Callable, List
from core.filters import FileFilter
from core.report import ReportGenerator
from core.compressor import Compressor

class ExportProjectService:
    def __init__(self, report_generator: ReportGenerator, compressor: Compressor):
        self.report_generator = report_generator
        self.compressor = compressor

    def execute(self, root_path: str, selected_paths: set, file_filter: FileFilter, progress_text_cb: Callable[[str], None], progress_val_cb: Callable[[int], None]):
        project_name = os.path.basename(os.path.normpath(root_path))
        
        # Gera o relatório com o nome do projeto (ex: domain-system.txt)
        named_report_path = os.path.join(root_path, f"{project_name}.txt")
        legacy_report_path = os.path.join(root_path, 'project.txt')
        zip_path = os.path.join(root_path, f"{project_name}.zip")

        # 1. Gerar Relatório de Contexto (0 a 50%)
        def report_progress(count: int, total: int, msg: str):
            percent = int((count / total) * 50) if total > 0 else 50
            progress_val_cb(percent)
            progress_text_cb(msg)

        progress_text_cb("Iniciando geração do relatório de contexto...")
        self.report_generator.generate(root_path, selected_paths, file_filter, named_report_path, report_progress)
        
        # Mantém project.txt sincronizado para ferramentas que esperam esse nome
        try:
            shutil.copy2(named_report_path, legacy_report_path)
        except Exception:
            pass

        # 2. Gerar ZIP do Projeto (50 a 100%)
        def zip_progress(count: int, total: int, msg: str):
            percent = 50 + int((count / total) * 50) if total > 0 else 100
            progress_val_cb(percent)
            progress_text_cb(f"Compactando: {msg}")

        progress_text_cb("Iniciando compactação em arquivo ZIP...")
        self.compressor.compress(root_path, selected_paths, file_filter, zip_path, zip_progress)
        
        progress_val_cb(100)
        progress_text_cb("Processo concluído com sucesso!")
        return True, f"Relatórios ({project_name}.txt e project.txt) e ZIP ({project_name}.zip) criados com sucesso!"

import os
from collections import defaultdict

class TreeFormatter:
    ICON_MAP = {
        '.py': '🐍',
        '.php': '🐘',
        '.js': '📜', '.jsx': '📜', '.ts': '📜', '.tsx': '📜',
        '.html': '🌐', '.htm': '🌐',
        '.css': '🎨', '.scss': '🎨', '.sass': '🎨',
        '.json': '📋',
        '.md': '📝', '.txt': '📝',
        '.yml': '⚙️', '.yaml': '⚙️',
        '.png': '🖼️', '.jpg': '🖼️', '.jpeg': '🖼️', '.svg': '🖼️', '.gif': '🖼️',
        '.sh': '🖥️', '.bat': '🖥️',
        '.env': '🔐',
        '.xml': '🔖', '.sql': '🗄️'
    }
    DEFAULT_ICON = '📄'

    @classmethod
    def _get_icon_for_extension(cls, filename: str) -> str:
        """Retorna um emoji correspondente à extensão do arquivo."""
        ext = os.path.splitext(filename)[1].lower()
        return cls.ICON_MAP.get(ext, cls.DEFAULT_ICON)

    @classmethod
    def get_tree_string(cls, root_path: str, selected_paths: set, progress_callback=None) -> str:
        """Gera a representação textual hierárquica completa da árvore de arquivos e pastas."""
        tree_lines = []
        project_name = os.path.basename(os.path.normpath(root_path))
        
        # Mapeia estrutura hierárquica completa: parent_rel_path -> set de (nome, rel_path, is_dir)
        children_map = defaultdict(dict)
        
        for p in selected_paths:
            rel = os.path.relpath(p, root_path).replace("\\", "/")
            if rel == '.' or rel == '':
                continue
            
            parts = rel.split('/')
            
            # Adiciona os diretórios ancestrais na hierarquia
            current_parent = ""
            for i in range(len(parts) - 1):
                dir_name = parts[i]
                dir_rel = f"{current_parent}/{dir_name}" if current_parent else dir_name
                children_map[current_parent][dir_name] = (dir_name, dir_rel, True)
                current_parent = dir_rel
            
            # Adiciona o arquivo (ou diretório folha)
            leaf_name = parts[-1]
            is_dir = os.path.isdir(p)
            children_map[current_parent][leaf_name] = (leaf_name, rel, is_dir)
        
        def build_tree(current_rel_path: str, prefix: str):
            items_dict = children_map.get(current_rel_path, {})
            # Converte para lista e ordena: primeiro pastas, depois arquivos (ordem alfabética)
            items = list(items_dict.values())
            items.sort(key=lambda x: (not x[2], x[0].lower()))
            
            for i, (name, rel_path, is_dir) in enumerate(items):
                if progress_callback:
                    progress_callback(1, 2, f"Formatando árvore: {name}")
                is_last = (i == len(items) - 1)
                connector = "└── " if is_last else "├── "
                
                if is_dir:
                    tree_lines.append(f"{prefix}{connector}📁 {name}/")
                    extension_prefix = "    " if is_last else "│   "
                    build_tree(rel_path, prefix + extension_prefix)
                else:
                    icon = cls._get_icon_for_extension(name)
                    tree_lines.append(f"{prefix}{connector}{icon} {name}")
                    
        tree_lines.append(f"📦 {project_name}/")
        build_tree("", "")
        return "\n".join(tree_lines)

class CheckableFileSystemModel(QFileSystemModel):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.checked_paths = set()

    def flags(self, index):
        default_flags = super().flags(index)
        if index.column() == 0:
            return default_flags | Qt.ItemIsUserCheckable
        return default_flags

    def data(self, index, role=Qt.DisplayRole):
        if role == Qt.CheckStateRole and index.column() == 0:
            path = self.filePath(index)
            return Qt.Checked if path in self.checked_paths else Qt.Unchecked
        return super().data(index, role)

    def setData(self, index, value, role=Qt.EditRole):
        if role == Qt.CheckStateRole and index.column() == 0:
            path = self.filePath(index)
            # PySide6 sends integer or Qt.CheckState enum
            is_checked = (value == Qt.Checked.value) if hasattr(value, 'value') else (value == 2)
            
            if is_checked:
                self.checked_paths.add(path)
            elif path in self.checked_paths:
                self.checked_paths.remove(path)
            
            self.dataChanged.emit(index, index, [Qt.CheckStateRole])
            return True
        return super().setData(index, value, role)

class ExceptionFilterProxyModel(QSortFilterProxyModel):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.ignore_patterns = []

    def set_ignore_patterns(self, patterns):
        self.ignore_patterns = patterns
        self.invalidateFilter()

    def filterAcceptsRow(self, source_row, source_parent):
        source_model = self.sourceModel()
        index = source_model.index(source_row, 0, source_parent)
        file_name = source_model.fileName(index)
        
        for pattern in self.ignore_patterns:
            if fnmatch.fnmatch(file_name.lower(), pattern.lower()):
                return False
        return True

            if name:

                added_count += 1

    def load_rules(self):
        for rw, txt, chk in self.ignore_rows:
            rw.deleteLater()
        self.ignore_rows.clear()

        rules_to_load = []
        if os.path.exists(CONFIG_FILE):
            try:
                with open(CONFIG_FILE, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    rules_to_load = data.get("rules", [])
            except Exception:
                rules_to_load = [{"pattern": d, "active": True} for d in (DEFAULT_IGNORE_DIRS + DEFAULT_IGNORE_FILES)]
        else:
            rules_to_load = [{"pattern": d, "active": True} for d in (DEFAULT_IGNORE_DIRS + DEFAULT_IGNORE_FILES)]

        for rule in rules_to_load:
            self.add_exception_row(rule["pattern"], rule["active"])

        if not rules_to_load:
            self.add_exception_row("", True)

    def save_rules(self):
        rules = []
        for rw, txt, chk in self.ignore_rows:
            pattern = txt.text().strip()
            if pattern:
                rules.append({"pattern": pattern, "active": chk.isChecked()})
        
        try:
            with open(CONFIG_FILE, 'w', encoding='utf-8') as f:
                json.dump({"rules": rules}, f, indent=4)
            QMessageBox.information(self, "Sucesso", "Regras salvas com sucesso!")
            self.rules_saved.emit()
        except Exception as e:
            QMessageBox.critical(self, "Erro", f"Erro ao salvar regras: {e}")

    def add_exception_row(self, initial_text="", is_active=True):
        row_widget = QWidget()
        row_layout = QHBoxLayout(row_widget)
        row_layout.setContentsMargins(0, 0, 0, 0)
        
        txt_field = QLineEdit()
        txt_field.setPlaceholderText("Ex: *.zip, docs, .git")
        txt_field.setText(initial_text)
        txt_field.setFixedWidth(300)
        
        chk_active = QCheckBox("Ativo")
        chk_active.setChecked(is_active)
        chk_active.stateChanged.connect(lambda: self._update_proxy_filters())
        txt_field.textChanged.connect(lambda: self._update_proxy_filters())
        
        btn_remove = QPushButton("Excluir")
        btn_remove.setFixedWidth(100)
        
        row_layout.addWidget(txt_field)
        row_layout.addWidget(chk_active)
        row_layout.addWidget(btn_remove)
        row_layout.addStretch()
        
        self.scroll_layout.addWidget(row_widget)
        
        btn_remove.clicked.connect(lambda: self.remove_exception_row(row_widget))
        self.ignore_rows.append((row_widget, txt_field, chk_active))
        self._update_proxy_filters()

    def remove_exception_row(self, row_widget):
        for i, (rw, txt, chk) in enumerate(self.ignore_rows):
            if rw == row_widget:
                self.ignore_rows.pop(i)
                rw.deleteLater()
                self._update_proxy_filters()
                break

    def _update_proxy_filters(self):
        if hasattr(self, 'proxy_model'):
            self.proxy_model.set_ignore_patterns(self.get_ignore_patterns())

    def get_ignore_patterns(self):
        patterns = []
        for rw, txt, chk in self.ignore_rows:
            if not chk.isChecked():
                continue
            t = txt.text().strip()
            if t:
                patterns.append(t)
        return patterns

        """Propaga o check/uncheck para os filhos e atualiza o pai se necessário."""
        self.tree_model.itemChanged.disconnect(self._on_item_changed)
        
        state = item.checkState()
        
        # Propagar para baixo
        def set_children_state(parent: QStandardItem, s: Qt.CheckState):
            for row in range(parent.rowCount()):
                child = parent.child(row)
                child.setCheckState(s)
                set_children_state(child, s)
        
        set_children_state(item, state)
        
        # Propagar para cima (Opcional: se todos os filhos desmarcados, desmarca o pai)
        # Para manter a simplicidade e performance, vamos apenas propagar para baixo.
        
        self.tree_model.itemChanged.connect(self._on_item_changed)

            # Mesmo se o pai estiver desmarcado, ele pode ter sido PartiallyChecked, mas 
            # usando 2-states é mais simples. Vamos iterar sobre os filhos.
            for row in range(item.rowCount()):
                collect_checked(item.child(row))
                
        root_item = self.tree_model.item(0, 0)
        if root_item:
            collect_checked(root_item)

        if not selected_paths:
            QMessageBox.warning(self, "Aviso", "Nenhum arquivo ou pasta selecionado!")
            return

import os
from PySide6.QtCore import QThread, Signal
from typing import List

from core.scanner import Scanner
from core.models import FileNode
from core.compressor import Compressor
from core.report import ReportGenerator

class ScanWorker(QThread):
    progress = Signal(str)
    item_found = Signal(str, str, bool, str)
    finished = Signal(FileNode)

    def __init__(self, scanner):
        super().__init__()
        self.scanner = scanner

    def run(self):
        try:
            self.progress.emit("Iniciando varredura...")
            
            def prog_cb(msg):
                self.progress.emit(msg)
                
            def item_cb(parent_path, full_path, is_dir, name):
                self.item_found.emit(parent_path, full_path, is_dir, name)
                
            root_node = self.scanner.scan(prog_cb, item_cb)
            self.finished.emit(root_node)
        except Exception as e:
            self.progress.emit(f"Erro na varredura: {str(e)}")
            self.finished.emit(None)


class ProcessWorker(QThread):
    progress_text = Signal(str)
    progress_value = Signal(int)
    finished = Signal(bool, str) # success, message

    def __init__(self, export_service, root_path: str, selected_paths: list, file_filter):
        super().__init__()
        self.export_service = export_service
        self.root_path = root_path
        self.selected_paths = set(selected_paths)
        self.file_filter = file_filter

    def run(self):
        try:
            self.progress_value.emit(0)
            
            def text_cb(msg): self.progress_text.emit(msg)
            def val_cb(val): self.progress_value.emit(val)

            success, msg = self.export_service.execute(
                self.root_path, 
                self.selected_paths, 
                self.file_filter, 
                text_cb, 
                val_cb
            )
            self.finished.emit(success, msg)
        except Exception as e:
            self.finished.emit(False, f"Erro crítico: {str(e)}")
